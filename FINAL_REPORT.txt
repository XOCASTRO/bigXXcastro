   $ node scripts/init-db.js

2. CREATE CONFIGURATION

   MYSQL_USER=root
   MYSQL_PASSWORD=your_password
   MYSQL_DATABASE=clinic_db

3. INSTALL & RUN
   $ pnpm install
   $ pnpm dev
